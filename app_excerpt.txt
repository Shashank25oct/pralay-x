function App() {
  const [screen, setScreen] = useState<'intro' | 'login' | 'public' | 'authority'>('intro');
  const [eventTick, setEventTick] = useState(0);
  const [auth, setAuth] = useState(!!localStorage.getItem('pralay_token'));

  useEffect(() => {
    const es = new EventSource(eventsUrl);
    const bump = () => setEventTick(v => v + 1);
    ['incident.created', 'incident.updated', 'complaint.created', 'complaint.updated', 'risk.updated', 'alert.created'].forEach(e => es.addEventListener(e, bump));
    return () => es.close();
  }, []);

  if (screen === 'intro') return <Intro onPublic={() => setScreen('public')} onLogin={() => setScreen('login')} />;
  if (screen === 'login') return <Login onBack={() => setScreen('intro')} onSuccess={() => { setAuth(true); setScreen('authority'); }} />;
  if (screen === 'authority' && auth) return <Authority tick={eventTick} onLogout={() => { localStorage.removeItem('pralay_token'); setAuth(false); setScreen('intro'); }} />;
  return <PublicSafety tick={eventTick} onAuthority={() => setScreen(auth ? 'authority' : 'login')} onHome={() => setScreen('intro')} />;
}

  const [busy, setBusy] = useState(false);

    e.preventDefault();

function ComplaintModal({ onClose }: { onClose: () => void }) {

  const [location, setLocation] = useState('Raipur, Chhattisgarh');

  const [done, setDone] = useState('');

  const submit = async (e: React.FormEvent) => {

function QuickActionModal({ onClose, onSimulation, onReport, onAI, onRefresh }: { onClose: () => void; onSimulation: () => void; onReport: () => void; onAI: () => void; onRefresh: () => void }) {
  const [action, setAction] = useState<string | null>(null);
  const [busy, setBusy] = useState(false);
  const [done, setDone] = useState('');
  const [state, setState] = useState('Chhattisgarh');
  const [district, setDistrict] = useState('Raipur');
  const [type, setType] = useState('Flood');
  const [severity, setSeverity] = useState('HIGH');
  const [title, setTitle] = useState('');
  const [message, setMessage] = useState('');
  const [resource, setResource] = useState('Rescue Team');
  const [quantity, setQuantity] = useState(1);

  const choose = (next: string) => { setDone(''); setAction(next); };
  const submit = async (e: React.FormEvent) => {
    e.preventDefault(); if (busy) return; setBusy(true); setDone('');
    try {
      if (action === 'CREATE_ALERT') {
        const d = await api.createAlert({ title: title || `${type} Alert — ${district}`, severity, state, district, message: message || `Authority-issued ${type.toLowerCase()} alert for ${district}.` });
        setDone(`Alert ${d.id} created successfully.`);
      } else if (action === 'REPORT_INCIDENT') {
        const d = await api.reportIncident({ disaster_type: type, severity, state, district, latitude: (stateDefaults[state] || stateDefaults['Chhattisgarh']).latitude, longitude: (stateDefaults[state] || stateDefaults['Chhattisgarh']).longitude, description: message || `Authority-reported ${type.toLowerCase()} incident in ${district}.`, affected_population: 0 });
        setDone(`Incident ${d.incident_id} recorded and sent for review.`);
      } else if (action === 'ADD_DATA') {
        const d = await api.addData({ disaster_type: type, severity, state, district, latitude: (stateDefaults[state] || stateDefaults['Chhattisgarh']).latitude, longitude: (stateDefaults[state] || stateDefaults['Chhattisgarh']).longitude, description: message || `New operational data entry for ${district}.` });
        setDone(`Data entry ${d.incident_id} added as UNVERIFIED.`);
      } else if (action === 'DISPATCH_RESOURCE') {
        const d = await api.dispatchResource({ resource_type: resource, quantity: Number(quantity), state, district, note: message });
        setDone(`Dispatch ${d.dispatch_id} created: ${quantity} × ${resource}.`);
      } else if (action === 'BROADCAST') {
        const d = await api.broadcast({ title: title || 'Authority Broadcast', severity, state, district, message: message || 'Official operational communication from Pralay X Command Centre.' });
        setDone(`Broadcast ${d.id} published to the authority/public alert stream.`);
      }
      onRefresh();
    } catch (e: any) { alert(e.message || 'Unable to complete this action.'); }
    finally { setBusy(false); }
  };

  const actions = [
    ['REPORT_INCIDENT', 'Report Incident', 'Create a new incident record for authority review.', <AlertTriangle size={20} />],
    ['CREATE_ALERT', 'Create Alert', 'Issue a warning or advisory for a selected area.', <Bell size={20} />],
    ['DISPATCH_RESOURCE', 'Dispatch Resource', 'Record deployment of rescue or emergency resources.', <Truck size={20} />],
    ['ADD_DATA', 'Add Data', 'Add a new operational data point for verification.', <Database size={20} />],
    ['GENERATE_REPORT', 'Generate Report', 'Generate the latest situation report.', <FileText size={20} />],
    ['OPEN_AI', 'Open AI', 'Jump to PRALAY AI Intelligence.', <Bot size={20} />],
    ['BROADCAST', 'Broadcast Message', 'Publish an official command-centre communication.', <Megaphone size={20} />],
  ] as const;


  return <Modal onClose={onClose} wide>
    <Badge type="info">COMMAND CENTRE</Badge><h2>Quick Action</h2><p>Execute a common operational action without leaving the command surface.</p>
    {!action ? <div className="quick-action-grid">{actions.map(([id, label, detail, icon]) => <button key={id} className="quick-action-card" onClick={() => { if (id === 'GENERATE_REPORT') { onReport(); onClose(); } else if (id === 'OPEN_AI') { onAI(); onClose(); } else if (id === 'REPORT_INCIDENT') { choose(id); } else if (id === 'CREATE_ALERT') { choose(id); } else if (id === 'DISPATCH_RESOURCE') { choose(id); } else if (id === 'ADD_DATA') { choose(id); } else if (id === 'BROADCAST') { choose(id); } }}><span className="quick-action-icon">{icon}</span><span><b>{label}</b><small>{detail}</small></span><ChevronRight size={16} /></button>)}</div> : <form onSubmit={submit}>
      <button type="button" className="quick-back" onClick={() => setAction(null)}>← ALL QUICK ACTIONS</button>
      <div className="quick-action-selected"><span className="quick-action-icon">{actions.find(a => a[0] === action)?.[3]}</span><div><b>{actions.find(a => a[0] === action)?.[1]}</b><small>{actions.find(a => a[0] === action)?.[2]}</small></div></div>
      <div className="form-grid">
        {action !== 'DISPATCH_RESOURCE' && <label>DISASTER / TYPE<select value={type} onChange={e => setType(e.target.value)}>{disasterOptions.map(x => <option key={x}>{x}</option>)}</select></label>}
        {action !== 'DISPATCH_RESOURCE' && <label>SEVERITY<select value={severity} onChange={e => setSeverity(e.target.value)}>{severityOptions.map(x => <option key={x}>{x}</option>)}</select></label>}
        <label>STATE<select value={state} onChange={e => { const next = e.target.value; setState(next); setDistrict(stateDefaults[next]?.district || 'Raipur'); }}>{stateOptions.map(x => <option key={x}>{x}</option>)}</select></label>
        <label>DISTRICT<input value={district} onChange={e => setDistrict(e.target.value)} /></label>
        {action === 'DISPATCH_RESOURCE' && <><label>RESOURCE<select value={resource} onChange={e => setResource(e.target.value)}>{['Rescue Team','Ambulance','Fire Unit','Boat','Helicopter','Medical Team','Relief Materials','Vehicle'].map(x => <option key={x}>{x}</option>)}</select></label><label>QUANTITY<input type="number" min="1" value={quantity} onChange={e => setQuantity(Number(e.target.value))} /></label></>}
        {(action === 'CREATE_ALERT' || action === 'BROADCAST') && <label>TITLE<input value={title} onChange={e => setTitle(e.target.value)} placeholder={action === 'BROADCAST' ? 'Official command message' : 'Alert title'} /></label>}
      </div>
      {action !== 'DISPATCH_RESOURCE' || true ? <label>{action === 'DISPATCH_RESOURCE' ? 'DISPATCH NOTE (OPTIONAL)' : action === 'BROADCAST' ? 'MESSAGE' : 'DESCRIPTION'}<textarea value={message} onChange={e => setMessage(e.target.value)} required={action === 'BROADCAST'} placeholder="Enter operational details…" /></label> : null}

      <div className="button-row"><button type="button" onClick={() => setAction(null)}>BACK</button><button className="primary" type="submit" disabled={busy}>{busy ? <><RefreshCw className="spin" size={16} /> PROCESSING…</> : <>EXECUTE ACTION <ArrowRight size={16} /></>}</button></div>
    </form>}
  </Modal>;
}
